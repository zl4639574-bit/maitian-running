/* ============================================================================
   麦田守望 · 成绩/资料收件中转（一份代码，通吃三个平台）
   ----------------------------------------------------------------------------
     · 腾讯云函数 SCF + API 网关触发器          → exports.main_handler
     · 腾讯云开发 CloudBase 云函数「HTTP 访问服务」→ exports.main
     · 阿里云函数计算 FC（HTTP 触发器/事件函数）  → exports.handler
     · 本地自测                                → exports.httpHandle(伪请求)

   做一件事：队员点「⚡ 直接提交给队长」→ POST 到这里 → 这里用令牌把内容写进
   仓库的 data/inbox/ → 队长端「收件箱」一键接收。

   环境变量（5 个，各平台都叫这些名字）：
     GH_TOKEN   写权限令牌（Contents: Read and write）—— 只放这里，绝不进网页
     GH_OWNER   zl4639574-bit
     GH_REPO    maitian-running
     GH_BRANCH  master
     TEAM_CODE  队伍口令（自己起）

   安全设计：只写 data/inbox/ 新文件（不带 sha⇒不能覆盖）、口令校验、同 IP 限流、
   字段白名单+长度上限、单次 ≤200KB（照片 ≤150KB）。
   ========================================================================== */

const GH = 'https://api.github.com';
const MAX_BYTES = 200 * 1024;
const MAX_PER_MIN = 5;
const hits = {};
const PB_LIST = ['800米', '1500米', '3000米', '5000米', '10000米', '半马', '全马'];

const corsHeaders = () => ({
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
  'Access-Control-Max-Age': '86400',
  'Content-Type': 'application/json; charset=utf-8',
});
const s = v => String(v == null ? '' : v).trim();
const badName = n => !/^[\u4e00-\u9fa5·a-zA-Z][\u4e00-\u9fa5·a-zA-Z0-9]{1,13}$/.test(s(n));

function validate(body, env) {
  if (s(body && body.code) !== s(env.TEAM_CODE)) return { ok: false, error: '队伍口令不对' };
  const type = s(body && body.type);
  if (type !== 'scores' && type !== 'member') return { ok: false, error: '类型只能是 scores / member' };
  const p = body && body.payload;
  if (!p || typeof p !== 'object') return { ok: false, error: '没有内容' };
  if (Buffer.byteLength(JSON.stringify(p), 'utf8') > MAX_BYTES) return { ok: false, error: '内容太大了（>200KB，照片请先压缩）' };

  if (type === 'member') {
    const name = s(p.name);
    if (badName(name)) return { ok: false, error: '姓名不合法' };
    const clean = { type: 'maitian-member', name,
      sex: s(p.sex).slice(0, 4), college: s(p.college).slice(0, 30),
      major: s(p.major).slice(0, 40), grade: s(p.grade).slice(0, 12), pb: {}, photo: '' };
    PB_LIST.forEach(k => { const v = s(p.pb && p.pb[k]); if (v && v !== '无') clean.pb[k] = v.slice(0, 20); });
    const ph = s(p.photo);
    if (/^data:image\/(jpeg|png|webp);base64,[A-Za-z0-9+/=]+$/.test(ph) && ph.length <= 150 * 1024) clean.photo = ph;
    return { ok: true, clean };
  }
  const rows = Array.isArray(p.rows) ? p.rows.slice(0, 200) : [];
  const cleanRows = [];
  rows.forEach(r => {
    const name = s(r && r.name);
    const fmt = s(r && r.fmt).slice(0, 20);
    if (badName(name) || !fmt) return;
    cleanRows.push({ name, event: s(r.event).slice(0, 20) || '5000米', fmt, sec: Number(r.sec) || 0,
      date: s(r.date).slice(0, 20), meet: s(r.meet).slice(0, 60), rank: s(r.rank).slice(0, 20) });
  });
  if (!cleanRows.length) return { ok: false, error: '没有可用的成绩行（每行要有姓名和成绩）' };
  return { ok: true, clean: { type: 'maitian-scores', date: s(p.date).slice(0, 20), rows: cleanRows } };
}

/** 核心：伪请求 → 伪响应（三个平台的适配器都调用它） */
async function httpHandle(req, env) {
  const method = s(req.method || 'POST').toUpperCase();
  if (method === 'OPTIONS') return { statusCode: 204, headers: corsHeaders(), body: '' };
  if (method !== 'POST') return { statusCode: 405, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '只接受 POST' }) };

  let body;
  try { body = JSON.parse(s(req.body) || '{}'); }
  catch (e) { return { statusCode: 400, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '提交内容不是合法 JSON' }) }; }

  const ip = s(req.headers && (req.headers['x-forwarded-for'] || req.headers['X-Forwarded-For'])).split(',')[0] || 'unknown';
  const now = Date.now();
  hits[ip] = (hits[ip] || []).filter(t => now - t < 60000);
  if (hits[ip].length >= MAX_PER_MIN) return { statusCode: 429, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '提交太频繁了，等一分钟再试' }) };
  hits[ip].push(now);

  const v = validate(body, env);
  if (!v.ok) return { statusCode: 400, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: v.error }) };
  if (!env.GH_TOKEN || !env.GH_OWNER || !env.GH_REPO)
    return { statusCode: 500, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '中转还没配好（缺 GH_TOKEN / GH_OWNER / GH_REPO）' }) };

  const id = new Date().toISOString().replace(/[-:T.Z]/g, '').slice(0, 14) + '-' + Math.random().toString(36).slice(2, 7);
  const path = 'data/inbox/' + id + '.json';
  const payload = JSON.stringify({ id, type: v.clean.type, at: new Date().toISOString(), data: v.clean }, null, 1);
  const url = GH + '/repos/' + env.GH_OWNER + '/' + env.GH_REPO + '/contents/' + path;

  for (let attempt = 0; attempt < 3; attempt++) {
    try {
      const r = await fetch(url, {
        method: 'PUT',
        headers: { Authorization: 'Bearer ' + env.GH_TOKEN, Accept: 'application/vnd.github+json',
          'Content-Type': 'application/json', 'X-GitHub-Api-Version': '2022-11-28', 'User-Agent': 'maitian-relay' },
        body: JSON.stringify({ message: '收到上报 ' + id + '（' + v.clean.type + '）',
          content: Buffer.from(payload, 'utf8').toString('base64'), branch: env.GH_BRANCH || 'master' }),
      });
      if (r.ok) return { statusCode: 200, headers: corsHeaders(), body: JSON.stringify({ ok: true, id }) };
      const t = await r.text();
      if (r.status === 401 || r.status === 403)
        return { statusCode: 500, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '中转的令牌没权限写仓库（Contents 要 Read and write）' }) };
      if (r.status === 422)
        return { statusCode: 409, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '这条已经收到过了，请勿重复提交' }) };
      if (attempt === 2)
        return { statusCode: 502, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '写仓库失败 ' + r.status + '：' + t.slice(0, 120) }) };
    } catch (e) {
      if (attempt === 2)
        return { statusCode: 502, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '连不上 GitHub：' + s(e && e.message).slice(0, 120) }) };
    }
    await new Promise(res => setTimeout(res, 800));
  }
  return { statusCode: 500, headers: corsHeaders(), body: JSON.stringify({ ok: false, error: '提交失败，稍后重试' }) };
}

/** 从各平台的 event 里取出 method / headers / body */
function fromEvent(event) {
  const e = event || {};
  const method = e.httpMethod || (e.requestContext && e.requestContext.httpMethod) ||
    (e.requestContext && e.requestContext.http && e.requestContext.http.method) || 'POST';
  const raw = e.isBase64Encoded ? Buffer.from(e.body || '', 'base64').toString('utf8') : (e.body || '{}');
  return { method, headers: e.headers || {}, body: raw };
}

/* ---------- 腾讯云函数 SCF（API 网关触发器）---------- */
exports.main_handler = async (event) => httpHandle(fromEvent(event), process.env);
/* ---------- 腾讯云开发 CloudBase（HTTP 访问服务 / 云接）---------- */
exports.main = async (event) => httpHandle(fromEvent(event), process.env);
/* ---------- 阿里云函数计算 FC ---------- */
exports.handler = async (event, context, callback) => {
  // 事件函数形态：event 里有 httpMethod/body；HTTP 函数形态：第一个参数是请求流
  if (event && typeof event.on === 'function') {
    const chunks = [];
    await new Promise((res) => { event.on('data', c => chunks.push(c)); event.on('end', res); });
    const req = { method: event.method, headers: event.headers || {}, body: Buffer.concat(chunks).toString('utf8') };
    const out = await httpHandle(req, process.env);
    if (context && context.res) {
      context.res.setHeader ? Object.keys(out.headers).forEach(k => context.res.setHeader(k, out.headers[k])) : null;
      context.res.statusCode = out.statusCode;
      context.res.send ? context.res.send(out.body) : callback(null, out.body);
      return;
    }
    return out;
  }
  return httpHandle(fromEvent(event), process.env);
};
/* ---------- 本地自测用 ---------- */
exports.httpHandle = httpHandle;
